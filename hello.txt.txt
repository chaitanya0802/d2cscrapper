    // 🔽 UNIVERSAL SCROLL METHOD — detects container scroll or window scroll
    public int scrollUntilLastProduct(WebDriver driver, Map<String, String> locators, String storeName, int timeoutSeconds) {
        JavascriptExecutor js = (JavascriptExecutor) driver;
        String productCardXpath = locators.get("productcard");
        String containerXpath = locators.get("productcardsection");

        long endTime = System.currentTimeMillis() + timeoutSeconds * 1000L;
        int prevCount = -1, stableCountTries = 0;
        final int STABLE_LIMIT = 5;
        final int SCROLL_INCREMENT = 600;
        final int PAUSE_MS = 1500;

        boolean useContainer = false;
        WebElement container = null;
        try {
            container = driver.findElement(By.xpath(containerXpath));
            Long scrollHeight = (Long) js.executeScript("return arguments[0].scrollHeight;", container);
            Long clientHeight = (Long) js.executeScript("return arguments[0].clientHeight;", container);
            if (scrollHeight != null && clientHeight != null && scrollHeight > clientHeight)
                useContainer = true;
        } catch (Exception ignored) {}

        prevCount = driver.findElements(By.xpath(productCardXpath)).size();
        System.out.println("Initial product count: " + prevCount + " | Container scroll: " + useContainer);

        String lastUrl = driver.getCurrentUrl();

        while (System.currentTimeMillis() < endTime) {
            try {
                if (useContainer && container != null)
                    js.executeScript("arguments[0].scrollTop = arguments[0].scrollTop + arguments[1];", container, SCROLL_INCREMENT);
                else
                    js.executeScript("window.scrollBy(0, arguments[0]);", SCROLL_INCREMENT);

                Thread.sleep(PAUSE_MS);

                // Handle ?pageCount= URL changes
                String currentUrl = driver.getCurrentUrl();
                if (!currentUrl.equals(lastUrl)) {
                    System.out.println("URL changed: " + currentUrl);
                    lastUrl = currentUrl;
                    Thread.sleep(1200);
                }

                int newCount = driver.findElements(By.xpath(productCardXpath)).size();
                if (newCount > prevCount) {
                    prevCount = newCount;
                    stableCountTries = 0;
                } else {
                    stableCountTries++;
                    if (stableCountTries >= STABLE_LIMIT) break;
                }

            } catch (Exception e) {
                System.out.println("Scroll exception: " + e.getMessage());
            }
        }

        System.out.println("Final loaded products: " + prevCount);
        return Math.max(prevCount, 0);
    }
