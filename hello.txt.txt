    public void scrapCategoryData(String subcat, String catURL, Map<String, String> locators) {
        List<Product> productList = new ArrayList<>();
        driver.navigate().to(catURL);

        // 🔽 NEW SCROLL METHOD
        int finalCount = scrollUntilLastProduct(driver, locators, storename, 120);
        Assert.assertNotEquals(finalCount, 0);

        List<WebElement> products = driver.findElements(By.xpath(locators.get("productcard")));
        System.out.println("===> Total products for (" + subcat + ") category are: " + products.size());

        for (int i = 0; i < products.size(); i++) {
            try {
                WebElement currproduct = products.get(i);
                WebDriverWait wait = new WebDriverWait(driver, Duration.ofSeconds(5));

                // ---- name ----
                String name = "";
                try {
                    name = currproduct.findElement(By.xpath(locators.get("name"))).getText().trim();
                } catch (Exception ignored) {}

                // ---- url ----
                String url = "";
                try {
                    if (locators.get("url").equalsIgnoreCase("routertype")) {
                        String clickXp = locators.get("imageurl");
                        String listingUrl = driver.getCurrentUrl();
                        WebElement clickTarget = currproduct.findElement(By.xpath(clickXp));
                        ((JavascriptExecutor) driver).executeScript("arguments[0].scrollIntoView({block:'center'});", clickTarget);
                        clickTarget.click();
                        wait.until(ExpectedConditions.not(ExpectedConditions.urlToBe(listingUrl)));
                        url = driver.getCurrentUrl();
                        driver.navigate().back();
                        wait.until(ExpectedConditions.urlToBe(listingUrl));
                        Thread.sleep(2000);
                        products = driver.findElements(By.xpath(locators.get("productcard")));
                        currproduct = products.get(i);
                    } else {
                        url = currproduct.findElement(By.xpath(locators.get("url"))).getAttribute("href");
                    }
                } catch (Exception ignored) {}

                if (name.isEmpty() || url.isEmpty()) continue;

                // ---- id ----
                String id = storename + Math.abs(url.hashCode());

                // ---- price ----
                int price = 0;
                try {
                    String pr = currproduct.findElement(By.xpath(locators.get("price"))).getText().trim();
                    price = (int) Math.round(Double.parseDouble(pr.replaceAll("[^0-9.]", "")));
                } catch (Exception ignored) {}

                // ---- discount_percent ----
                int discount_percent = 0;
                try {
                    String disp = currproduct.findElement(By.xpath(locators.get("discount_percent"))).getText().trim();
                    discount_percent = Math.round(Integer.parseInt(disp.replaceAll("[^0-9.]", "")));
                } catch (Exception ignored) {}

                // ---- imageurl ----
                String imageurl = "";
                try {
                    imageurl = currproduct.findElement(By.xpath(locators.get("imageurl"))).getAttribute("src");
                } catch (Exception ignored) {}

                // ---- description ----
                String des = "";
                try {
                    des = currproduct.findElement(By.xpath(locators.get("description"))).getText().trim();
                } catch (Exception ignored) {}

                // ---- rating ----
                float rating = 0f;
                try {
                    String rt = currproduct.findElement(By.xpath(locators.get("rating"))).getText().trim();
                    rating = Float.parseFloat(rt.replaceAll("[^0-9.]", ""));
                } catch (Exception ignored) {}

                // ---- availability ----
                boolean isAvailable = false;
                try {
                    String idf = locators.get("availablity_idf");
                    isAvailable = !currproduct.findElements(By.xpath(".//descendant::*[contains(text(), '" + idf + "')]")).isEmpty();
                } catch (Exception ignored) {}

                Product p = new Product(id, name, url, imageurl, maincat, subcat, price, discount_percent, des, rating, isAvailable);
                productList.add(p);
                System.out.println(p);
                System.out.println("=====================================");

            } catch (Exception e) {
                System.out.println("Error at product " + i + ": " + e.getMessage());
            }
        }
    }

    // 🔽 UNIVERSAL SCROLL METHOD — detects container scroll or window scroll
    public int scrollUntilLastProduct(WebDriver driver, Map<String, String> locators, String storeName, int timeoutSeconds) {
        JavascriptExecutor js = (JavascriptExecutor) driver;
        String productCardXpath = locators.get("productcard");
        String containerXpath = locators.get("productcardsection");

        long endTime = System.currentTimeMillis() + timeoutSeconds * 1000L;
        int prevCount = -1, stableCountTries = 0;
        final int STABLE_LIMIT = 5;
        final int SCROLL_INCREMENT = 600;
        final int PAUSE_MS = 1500;

        boolean useContainer = false;
        WebElement container = null;
        try {
            container = driver.findElement(By.xpath(containerXpath));
            Long scrollHeight = (Long) js.executeScript("return arguments[0].scrollHeight;", container);
            Long clientHeight = (Long) js.executeScript("return arguments[0].clientHeight;", container);
            if (scrollHeight != null && clientHeight != null && scrollHeight > clientHeight)
                useContainer = true;
        } catch (Exception ignored) {}

        prevCount = driver.findElements(By.xpath(productCardXpath)).size();
        System.out.println("Initial product count: " + prevCount + " | Container scroll: " + useContainer);

        String lastUrl = driver.getCurrentUrl();

        while (System.currentTimeMillis() < endTime) {
            try {
                if (useContainer && container != null)
                    js.executeScript("arguments[0].scrollTop = arguments[0].scrollTop + arguments[1];", container, SCROLL_INCREMENT);
                else
                    js.executeScript("window.scrollBy(0, arguments[0]);", SCROLL_INCREMENT);

                Thread.sleep(PAUSE_MS);

                // Handle ?pageCount= URL changes
                String currentUrl = driver.getCurrentUrl();
                if (!currentUrl.equals(lastUrl)) {
                    System.out.println("URL changed: " + currentUrl);
                    lastUrl = currentUrl;
                    Thread.sleep(1200);
                }

                int newCount = driver.findElements(By.xpath(productCardXpath)).size();
                if (newCount > prevCount) {
                    prevCount = newCount;
                    stableCountTries = 0;
                } else {
                    stableCountTries++;
                    if (stableCountTries >= STABLE_LIMIT) break;
                }

            } catch (Exception e) {
                System.out.println("Scroll exception: " + e.getMessage());
            }
        }

        System.out.println("Final loaded products: " + prevCount);
        return Math.max(prevCount, 0);
    }
